import logging
import json
import sys
from datetime import datetime

class JSONFormatter(logging.Formatter):
    def format(self, record):
        log_entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "name": record.name,
            "message": record.getMessage(),
        }
        if record.exc_info:
            log_entry["exception"] = self.formatException(record.exc_info)
        return json.dumps(log_entry)

def setup_logging(level=logging.INFO):
    handler = logging.StreamHandler(sys.stdout)
    handler.setFormatter(JSONFormatter())
    root = logging.getLogger()
    root.setLevel(level)
    root.addHandler(handler)
    return root

def retry(max_attempts=3, delay=2):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    time.sleep(delay * (2 ** attempt))
            return None
        return wrapper

python
import time
import threading
from collections import deque

class TokenBucket:
    def __init__(self, rpm: float, burst: int = 1, min_gap: float = 1.0):
        self.rpm = max(1.0, rpm)
        self.burst = burst
        self.min_gap = min_gap
        self.tokens = float(burst)
        self.last_update = time.monotonic()
        self.last_request = 0.0
        self.lock = threading.RLock()
        self.history = deque()

    def _refill(self):
        now = time.monotonic()
        elapsed = now - self.last_update
        self.tokens = min(self.burst, self.tokens + elapsed * (self.rpm / 60.0))
        self.last_update = now

    def _prune(self):
        cutoff = time.monotonic() - 60.0
        while self.history and self.history[0] < cutoff:
            self.history.popleft()

    def try_acquire(self) -> bool:
        with self.lock:
            self._refill()
            self._prune()
            if self.tokens < 1.0:
                return False
            if self.history and len(self.history) >= int(self.rpm):
                return False
            if time.monotonic() - self.last_request < self.min_gap:
                return False
            self.tokens -= 1.0
            self.last_request = time.monotonic()
            self.history.append(self.last_request)
            return True

    def wait_until_available(self, timeout: float = 5.0) -> bool:
        start = time.monotonic()
        while time.monotonic() - start < timeout:
            if self.try_acquire():
                return True
            time.sleep(0.1)
        return False

from typing import List, Optional, Dict, Any

class Provider:
    def __init__(self, name: str, client, model_id: str, rpm: float):
        self.name = name
        self.client = client
        self.model_id = model_id
        self.bucket = TokenBucket(rpm)
        self.cooldown_until = 0.0

    def is_available(self) -> bool:
        return time.monotonic() >= self.cooldown_until and self.bucket.try_acquire()

class ProviderPool:
    def __init__(self, providers: List[Provider]):
        self.providers = providers
        self.index = 0

    def next_available(self) -> Optional[Provider]:
        for _ in range(len(self.providers)):
            p = self.providers[self.index]
            if p.is_available():
                return p
            self.index = (self.index + 1) % len(self.providers)
        return None

    def mark_failed(self, provider: Provider, cooldown: float = 15.0):

from pathlib import Path
import os

class FileTools:
    def __init__(self, base: str):
        self.base = Path(base).resolve()

    def _safe(self, p: str) -> Path:
        t = (self.base / p).resolve()
        if not str(t).startswith(str(self.base)):
            raise PermissionError("Path outside project")
        return t

    def read(self, path: str) -> str:
        t = self._safe(path)
        with open(t, 'r', encoding='utf-8') as f:
            return f.read()

    def write(self, path: str, content: str) -> str:
        t = self._safe(path)
        t.parent.mkdir(parents=True, exist_ok=True)
        with open(t, 'w', encoding='utf-8') as f:
            f.write(content)
        return f"Written {len(content)} chars"

    def search(self, pattern: str, path: str = ".") -> str:
        import re
        t = self._safe(path)
        matches = []
        for root, dirs, files in os.walk(t):
            for f in files:
                fp = Path(root) / f
                try:
                    with open(fp, 'r', encoding='utf-8', errors='replace') as fh:
                        for i, line in enumerate(fh, 1):
                            if re.search(pattern, line):
                                matches.append(f"{fp.relative_to(self.base)}:{i}:{line.strip()}")
                except:
                    pass
        return "\n".join(matches)

def apply_regex_to_file(file_path, pattern, replacement, description):
    """Apply a regex substitution to a file. Returns True if changed."""
    if not file_path.exists():
        log_warning(f"{description} — file not found: {file_path}")
        return False
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        new_content, count = re.subn(pattern, replacement, content, flags=re.MULTILINE)
        if count > 0:
            with open(file_path, 'w', encoding='utf-8') as f:
                f.write(new_content)
            log_success(f"{description} — applied ({count} replacements)")
            return True
        else:
            log_warning(f"{description} — pattern not found")
            return False
    except Exception as e:
        log_error(f"{description} — error: {e}")
        return False
